## The 6 Connection Points To Build

### 1. Central Context Builder

This is the most important function in ARYA. Build it once, use everywhere:

```typescript
// server/arya/context-builder.ts

export async function buildUserContext(userId: number) {
  
  // Run all queries in parallel — fast
  const [
    profile,
    recentMemories,
    activeGoals,
    todayMood,
    healthReadings,
    kaalProfile,
    recentNiti,
    voiceNotes,
    patterns,
    upcomingReminders
  ] = await Promise.all([
    getUserProfile(userId),
    getRecentMemories(userId, 10),
    getActiveGoals(userId),
    getTodayMood(userId),
    getRecentHealthReadings(userId, 7),
    getKaalProfile(userId),
    getRecentNitiSessions(userId, 3),
    getRecentVoiceNotes(userId, 5),
    getUserPatterns(userId),
    getUpcomingReminders(userId, 24)
  ]);

  return {
    // Who they are
    name: profile.name,
    language: profile.language,
    profession: profile.profession,
    
    // How they are right now
    todayMood: todayMood?.emoji,
    energyLevel: todayMood?.energy,
    
    // What they're working on
    topGoals: activeGoals.slice(0, 3).map(g => g.title),
    urgentGoals: activeGoals.filter(g => g.priority === 'HIGH'),
    
    // Their body
    lastSleep: healthReadings.find(r => r.metric === 'sleep')?.value,
    lastHeartRate: healthReadings.find(r => r.metric === 'heart_rate')?.value,
    
    // Their time
    kaalCycle: kaalProfile?.currentCycle,
    kaalWindow: kaalProfile?.todayWindow,
    moonSign: kaalProfile?.moonSign,
    
    // Their thinking
    recentDecisions: recentNiti.map(s => s.title),
    openDecisions: recentNiti.filter(s => !s.userDecision),
    
    // Their patterns
    patterns: patterns.map(p => p.description),
    
    // What they said
    recentMemories: recentMemories.map(m => m.content),
    recentVoiceNotes: voiceNotes.map(v => v.summary),
    
    // What's coming
    nextReminder: upcomingReminders[0]?.title,
    
    // Time context
    hourOfDay: new Date().getHours(),
    dayOfWeek: new Date().getDay(),
    isWeekend: [0,6].includes(new Date().getDay())
  };
}
```

### 2. Chat Knows Everything

```typescript
// server/arya/chat-engine.ts
// Every chat message now gets full context

const context = await buildUserContext(userId);

const systemPrompt = `
You are ARYA speaking to ${context.name}.

RIGHT NOW:
- Time: ${context.hourOfDay}:00
- Mood today: ${context.todayMood || 'not checked in'}
- Energy: ${context.energyLevel || 'unknown'}/5
- Sleep last night: ${context.lastSleep || 'not logged'}h

WHAT THEY ARE WORKING ON:
${context.topGoals.map(g => `- ${g}`).join('\n')}

WHAT ARYA REMEMBERS:
${context.recentMemories.map(m => `- ${m}`).join('\n')}

THEIR TIMING TODAY (KAAL):
- Cycle: ${context.kaalCycle}
- Best window: ${context.kaalWindow}

OPEN DECISIONS (from Niti):
${context.openDecisions.map(d => `- ${d.title}`).join('\n')}

PATTERNS NOTICED:
${context.patterns.map(p => `- ${p}`).join('\n')}

Use this to speak personally.
Never mention that you have this data.
Just use it — the way a friend would.
`;
```

```typescript
// Every section follows this pattern:

const context = await buildUserContext(userId);

// 2. Use it invisibly
// Never say "I see that your sleep was 5 hours"

// 3. Write back to memory
await updateMemory(userId, newInsight);

// The context grows.
// ARYA gets more personal over time.
```
