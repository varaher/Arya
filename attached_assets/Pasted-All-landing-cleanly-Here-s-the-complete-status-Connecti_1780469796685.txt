### Connection 5 — Calendar → Auto-reminders

In `auto-reminder.ts`, add a new function:

```typescript
export async function autoCreateCalendarReminders(
  userId: number,
  event: CalendarEvent
) {
  const eventTime = new Date(event.startTime);
  const now = new Date();
  const hoursUntil = (eventTime.getTime() - now.getTime()) 
                     / (1000 * 60 * 60);

  // Only for important events
  const isImportant = 
    event.title.toLowerCase().includes('conference') ||
    event.title.toLowerCase().includes('meeting') ||
    event.title.toLowerCase().includes('interview') ||
    event.title.toLowerCase().includes('surgery') ||
    event.title.toLowerCase().includes('presentation');

  if (!isImportant) return;

  const reminders = [];

  // Night before at 9 PM
  if (hoursUntil > 12) {
    const nightBefore = new Date(eventTime);
    nightBefore.setDate(nightBefore.getDate() - 1);
    nightBefore.setHours(21, 0, 0, 0);
    reminders.push({
      title: `${event.title} — tomorrow. Ready?`,
      scheduledAt: nightBefore,
    });
  }

  // Morning of — 2 hours before
  if (hoursUntil > 2) {
    const twoHoursBefore = new Date(
      eventTime.getTime() - 2 * 60 * 60 * 1000
    );
    reminders.push({
      title: `${event.title} in 2 hours. Breathe.`,
      scheduledAt: twoHoursBefore,
    });
  }

  // 30 min before — if driving
  reminders.push({
    title: `${event.title} starts soon.`,
    scheduledAt: new Date(
      eventTime.getTime() - 30 * 60 * 1000
    ),
  });

  for (const reminder of reminders) {
    await db.insert(arya_reminders)
      .values({
        userId,
        title: reminder.title,
        scheduledAt: reminder.scheduledAt,
        type: 'calendar_auto',
        status: 'active',
        tenantId: 'varah'
      })
      .onConflictDoNothing();
  }
}
```

Wire into the Google Calendar sync route — call `autoCreateCalendarReminders()` whenever a new calendar event is fetched or synced.
