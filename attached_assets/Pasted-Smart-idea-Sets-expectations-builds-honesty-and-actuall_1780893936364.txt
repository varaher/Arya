## The Popup — What It Should Say

Not a warning. Not an apology. Just honest:

```
🎙️ Voice in [Language]

ARYA speaks and listens in [Malayalam].
This feature is in beta — it works well
for most conversations but may occasionally
mishear or mispronounce.

Your feedback makes it better.
[Got it]  [Report an issue]
```

---

## When To Show It

**Only once per language** — not every time:

```typescript
// Show popup when:
// 1. User selects an Indian language for the first time
// 2. User starts voice chat in that language
// 3. Never show again for that language after dismissed

// Storage key:
`arya_lang_beta_seen_${langCode}`
// e.g. "arya_lang_beta_seen_ml-IN"
```

## Exact Code — Add To AryaChat.tsx

```tsx
// ── State ──────────────────────────────────────────
const [betaLangPopup, setBetaLangPopup] = useState<{
  show: boolean;
  language: string;
  langCode: string;
} | null>(null);

// ── Language names ──────────────────────────────────
const LANG_NAMES: Record<string, string> = {
  'ml-IN': 'Malayalam',
  'hi-IN': 'Hindi',
  'ta-IN': 'Tamil',
  'te-IN': 'Telugu',
  'kn-IN': 'Kannada',
  'bn-IN': 'Bengali',
  'mr-IN': 'Marathi',
  'gu-IN': 'Gujarati',
  'pa-IN': 'Punjabi',
  'od-IN': 'Odia',
};

const INDIAN_LANG_CODES = Object.keys(LANG_NAMES);

// ── Show beta popup when language changes ───────────
function handleLanguageChange(newLangCode: string) {
  setSelectedLanguage(newLangCode);
  localStorage.setItem('arya_lang', newLangCode);

  // Only for Indian languages
  if (!INDIAN_LANG_CODES.includes(newLangCode)) return;

  // Only if user hasn't seen it for this language
  const seenKey = `arya_lang_beta_seen_${newLangCode}`;
  const alreadySeen = localStorage.getItem(seenKey);
  if (alreadySeen) return;

  // Show popup
  setBetaLangPopup({
    show: true,
    language: LANG_NAMES[newLangCode],
    langCode: newLangCode,
  });
}

// ── Dismiss ─────────────────────────────────────────
function dismissBetaPopup(reportIssue = false) {
  if (!betaLangPopup) return;

  // Mark as seen — never show again for this language
  localStorage.setItem(
    `arya_lang_beta_seen_${betaLangPopup.langCode}`,
    'true'
  );

  setBetaLangPopup(null);

  if (reportIssue) {
    // Open feedback — send a pre-filled message
    setInput(`Voice feedback for ${betaLangPopup.language}: `);
  }
}
```

## The Popup Component

```tsx
{betaLangPopup?.show && (
  <div
    className="beta-popup-overlay"
    onClick={() => dismissBetaPopup()}
  >
    <div
      className="beta-popup"
      onClick={e => e.stopPropagation()}
    >
      {/* Header */}
      <div className="beta-popup-header">
        <span className="beta-popup-icon">🎙️</span>
        <div>
          <p className="beta-popup-title">
            Voice in {betaLangPopup.language}
          </p>
          <span className="beta-badge">BETA</span>
        </div>
      </div>

      {/* Body */}
      <p className="beta-popup-body">
        ARYA speaks and listens in{' '}
        <strong>{betaLangPopup.language}</strong>.
        This feature works well for most conversations
        but may occasionally mishear or mispronounce.
      </p>

      <p className="beta-popup-sub">
        Your feedback makes it better for everyone.
      </p>

      {/* Actions */}
      <div className="beta-popup-actions">
        <button
          className="beta-btn-report"
          onClick={() => dismissBetaPopup(true)}
        >
          Report an issue
        </button>
        <button
          className="beta-btn-got-it"
          onClick={() => dismissBetaPopup()}
        >
          Got it →
        </button>
      </div>
    </div>
  </div>
)}
```
