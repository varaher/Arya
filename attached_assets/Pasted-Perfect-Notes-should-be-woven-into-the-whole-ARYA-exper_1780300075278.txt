## The Notification Flow — Exact Logic

```typescript
// server/arya/notes-scheduler.ts
// Run daily at 8 AM

async function checkNoteReminders(userId: string) {

  // Get all notes with unactioned tasks
  const notes = await db.query(`
    SELECT * FROM arya_voice_notes
    WHERE user_id = $1
    AND tasks_saved_to_goals = false
    AND created_at >= NOW() - INTERVAL '7 days'
  `, [userId]);

  if (notes.length > 0) {
    const taskCount = notes.reduce((sum, note) =>
      sum + (note.extracted_tasks?.length || 0), 0
    );

    if (taskCount > 0) {
      await sendNotification(userId, {
        type: 'notes_reminder',
        title: '🎙️ Unactioned voice notes',
        body: `You have ${taskCount} items from
               your notes not yet added to goals.`,
        data: { screen: 'notes' }
      });
    }
  }

  // Check deadline reminders
  const deadlineNotes = await db.query(`
    SELECT * FROM arya_voice_notes
    WHERE user_id = $1
    AND extracted_deadlines != '[]'::jsonb
    AND created_at >= NOW() - INTERVAL '7 days'
  `, [userId]);

  for (const note of deadlineNotes) {
    for (const deadline of note.extracted_deadlines) {
      const dueDate = new Date(deadline.date);
      const tomorrow = new Date();
      tomorrow.setDate(tomorrow.getDate() + 1);

      if (dueDate.toDateString() ===
          tomorrow.toDateString()) {
        await sendNotification(userId, {
          type: 'deadline_reminder',
          title: '📅 Due tomorrow',
          body: `From your note: ${deadline.task}`,
          data: { screen: 'notes', noteId: note.id }
        });
      }
    }
  }
}
```
