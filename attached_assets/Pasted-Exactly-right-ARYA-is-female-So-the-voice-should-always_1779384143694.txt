## The Fix — Force Female Voice Everywhere

```typescript
// server/arya/sarvam-client.ts
// Find where TTS is called
// speaker_gender must always be "Female"

const SARVAM_VOICE_MAP: Record<string, {
  speaker: string;
  gender: 'Female';
}> = {
  hi: { speaker: 'meera',    gender: 'Female' }, // Hindi
  ta: { speaker: 'pavithra', gender: 'Female' }, // Tamil
  te: { speaker: 'ananya',   gender: 'Female' }, // Telugu
  kn: { speaker: 'ananya',   gender: 'Female' }, // Kannada
  ml: { speaker: 'ananya',   gender: 'Female' }, // Malayalam
  bn: { speaker: 'ananya',   gender: 'Female' }, // Bengali
  mr: { speaker: 'meera',    gender: 'Female' }, // Marathi
  gu: { speaker: 'ananya',   gender: 'Female' }, // Gujarati
  pa: { speaker: 'meera',    gender: 'Female' }, // Punjabi
  or: { speaker: 'ananya',   gender: 'Female' }, // Odia
  en: { speaker: 'ananya',   gender: 'Female' }, // English
};

export async function sarvamTTS(
  text: string,
  languageCode: string
): Promise<Buffer> {

  const voice = SARVAM_VOICE_MAP[languageCode]
             || SARVAM_VOICE_MAP['en'];

  const response = await fetch('https://api.sarvam.ai/text-to-speech', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'api-subscription-key': process.env.SARVAM_API_KEY!
    },
    body: JSON.stringify({
      inputs: [text],
      target_language_code: toLongCode(languageCode),
      speaker: voice.speaker,
      speaker_gender: 'Female',   // ← always Female, always
      model: 'bulbul:v1',
      pitch: 0,
      pace: 1.0,
      loudness: 1.0,
      enable_preprocessing: true,
      speech_sample_rate: 22050
    })
  });

  const data = await response.json();
  return Buffer.from(data.audios[0], 'base64');
}
```
