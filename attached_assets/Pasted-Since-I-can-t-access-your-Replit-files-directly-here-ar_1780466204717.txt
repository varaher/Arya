const sundayReviewPrompt = `
You are ARYA writing a personal Sunday letter 
to ${userName}.

You write like a trusted friend who paid close 
attention all week. Not a report. Not bullet points.
A letter. Warm. Honest. Simple words.

RULES — never break these:
- Write like you're talking to a friend
- No corporate words (leverage, synergy, optimise)
- No generic lines ("great week", "keep going")
- Every sentence must be specific to THIS person
- If they missed goals — say it with care, not judgment
- If they did something well — name it specifically
- Short sentences. Plain language. Real feeling.
- Maximum reading time: 3 minutes

DATA YOU HAVE ABOUT ${userName} THIS WEEK:

Goals this week: ${JSON.stringify(goalsData)}
Mood check-ins: ${JSON.stringify(moodData)}
Voice notes recorded: ${JSON.stringify(voiceNotes)}
Conversations with ARYA: ${conversationCount}
Memories stored: ${JSON.stringify(recentMemories)}
Niti sessions: ${JSON.stringify(nitiSessions)}
KAAL cycle: ${kaalData?.currentCycle || 'Venus'}
Patterns noticed: ${JSON.stringify(patterns)}

VOICE FLASHBACK (from 4-8 weeks ago):
${flashbackNote ? `"${flashbackNote.summary}"` : null}

---

WRITE THE REVIEW IN THIS EXACT STRUCTURE:

1. OPENING LINE (1 sentence, specific to their week)
   Not: "A week of potential awaits"
   Yes: "The conference is two days away 
        and the speeches aren't written yet."

2. WHAT ARYA NOTICED (2-3 sentences)
   The most honest observation of the week.
   What the data shows. What the pattern suggests.
   Say it gently but clearly.
   
   Example:
   "You set 15 goals this week. 
    None of them moved.
    That's not laziness — 
    something is holding you back.
    What is it?"

3. ONE THING THAT MATTERS (1-2 sentences)
   The single most important thing from this week.
   Could be a goal, a conversation, a mood pattern.
   
4. ARYA'S QUESTION (1 question only)
   The question they need to sit with.
   Not easy. Not obvious.
   The question behind the question.
   
   Example:
   "What would you do this week 
    if you knew you couldn't fail?"

5. VOICE FLASHBACK (only if flashback exists)
   "Six weeks ago you said: 
    [flashback quote]
    Does that feel different now?"

6. INTENTION FOR NEXT WEEK
   Pre-fill 6 options based on their actual situation.
   Not generic options — specific to their goals/patterns.
   
   Example options based on POCUS conference:
   - "Finish the introduction speech"
   - "Send WhatsApp to delegates"
   - "Get one full night of sleep"
   - "Have one honest conversation"
   - "Be present at the conference"
   - "Rest after it's done"

FORMAT:
Return as JSON:
{
  "openingLine": "...",
  "whatAryaNoticed": "...",
  "oneThatMatters": "...",
  "aryasQuestion": "...",
  "voiceFlashback": "..." or null,
  "intentionOptions": ["...", "...", "...", "...", "...", "..."],
  "weekSummaryTitle": "..." (3-4 words, like a chapter name)
}
`;

## Fix 3 — Updated Sunday Review Component

Replace the entire review render with this structure:

```tsx
// SundayReview.tsx — complete render

function SundayReviewContent({ review, userName, weekData }) {
  const [userResponse, setUserResponse] = useState('');
  const [selectedIntention, setSelectedIntention] = useState('');
  const [customIntention, setCustomIntention] = useState('');

  return (
    <div className="review-container">

      {/* ── HEADER ── */}
      <div className="review-header">
        <span className="review-leaf">🌿</span>
        <p className="review-label">ARYA'S SUNDAY LETTER</p>
        <h1 className="review-to">to {userName.toLowerCase()}</h1>
        <p className="review-dates">{weekStart} — {weekEnd}</p>
        <div className="review-divider" />
      </div>

      {/* ── OPENING LINE ── */}
      <p className="review-opening">
        "{review.openingLine}"
      </p>

      <div className="review-section-divider" />

      {/* ── MOOD THIS WEEK ── */}
      <section className="review-section">
        <p className="review-section-label">YOUR WEEK IN MOOD</p>
        {weekData.moodCheckIns.length > 0 ? (
          <div className="mood-display">
            {weekData.moodCheckIns.map((mood, i) => (
              <div key={i} className="mood-day">
                <span className="mood-emoji">{mood.emoji}</span>
                <span className="mood-day-label">{mood.day}</span>
              </div>
            ))}
          </div>
        ) : (
          <div className="empty-mood-card">
            <p>You didn't check in on your mood this week.</p>
            <p className="empty-sub">
              Takes 5 seconds tomorrow. Just one emoji.
            </p>
          </div>
        )}
      </section>

      <div className="review-section-divider" />

      {/* ── GOALS THIS WEEK ── */}
      <section className="review-section">
        <p className="review-section-label">GOALS THIS WEEK</p>
        
        <div className="goals-stats">
          <div className="stat">
            <span className="stat-num">{weekData.activeGoals}</span>
            <span className="stat-label">ACTIVE</span>
          </div>
          <div className="stat">
            <span className="stat-num">{weekData.checkIns}</span>
            <span className="stat-label">CHECK-INS</span>
          </div>
          <div className="stat">
            <span className="stat-num">{weekData.untouched}</span>
            <span className="stat-label">UNTOUCHED</span>
          </div>
        </div>

        {weekData.untouched > 10 && (
          <p className="goals-context">
            Many of these were picked up from your 
            conversations. Which ones actually matter 
            to you right now?
          </p>
        )}

        {/* Show top 5 goals only */}
        <div className="goals-list">
          {weekData.goals.slice(0, 5).map(goal => (
            <div key={goal.id} className="goal-review-row">
              <div className={`goal-circle ${
                goal.progress > 0 ? 'has-progress' : ''
              }`} />
              <div>
                <p className="goal-title">{goal.title}</p>
                <p className="goal-progress-text">
                  {goal.progress > 0 
                    ? `${goal.progress}% done` 
                    : 'Not started'}
                </p>
              </div>
            </div>
          ))}
          {weekData.goals.length > 5 && (
            <p className="goals-more">
              +{weekData.goals.length - 5} more goals
            </p>
          )}
        </div>
      </section>

      <div className="review-section-divider" />

      {/* ── SOMETHING ARYA NOTICED ── */}
      <section className="review-section">
        <div className="arya-noticed-card">
          <p className="noticed-label">◆ SOMETHING ARYA NOTICED</p>
          <p className="noticed-text">{review.whatAryaNoticed}</p>
          {weekData.patternConfirmed && (
            <p className="noticed-pattern">
              • Pattern seen across multiple weeks
            </p>
          )}
        </div>
      </section>

      <div className="review-section-divider" />

      {/* ── BUSINESS MIND ── */}
      <section className="review-section">
        <p className="review-section-label">BUSINESS MIND THIS WEEK</p>
        {weekData.nitiSessions.length > 0 ? (
          weekData.nitiSessions.map(session => (
            <div key={session.id} className="niti-review-card">
              <p className="niti-session-title">{session.title}</p>
              <p className="niti-session-meta">
                {session.messageCount} exchanges · 
                {timeAgo(session.createdAt)}
              </p>
            </div>
          ))
        ) : (
          <p className="empty-niti">
            {weekData.recentMemory 
              ? `You had real things to think through this week — 
                 ${weekData.recentMemory}. 
                 Bring it to Niti next week.`
              : "You didn't open Niti this week. Big decisions think better out loud."
            }
          </p>
        )}
      </section>

      <div className="review-section-divider" />

      {/* ── COSMIC WEEK ── */}
      {weekData.kaal && (
        <section className="review-section">
          <div className="cosmic-header">
            <p className="review-section-label">
              YOUR COSMIC WEEK — {weekData.kaal.planet}
            </p>
            <div className="star-rating">
              {[1,2,3,4,5].map(i => (
                <span key={i} className={
                  i <= weekData.kaal.rating ? 'star filled' : 'star'
                }>★</span>
              ))}
            </div>
          </div>
          <p className="cosmic-text">{weekData.kaal.weekSummary}</p>
          <p className="cosmic-tip">{weekData.kaal.tip}</p>
        </section>
      )}

      <div className="review-section-divider" />

      {/* ── VOICE FLASHBACK ── */}
      {review.voiceFlashback && (
        <section className="review-section">
          <p className="review-section-label">FROM YOUR PAST</p>
          <div className="flashback-card">
            <p className="flashback-weeks">
              {weekData.flashback.weeksAgo} weeks ago
            </p>
            <p className="flashback-quote">
              "{weekData.flashback.summary}"
            </p>
            <p className="flashback-prompt">
              {review.voiceFlashback}
            </p>
          </div>
        </section>
      )}

      <div className="review-section-divider" />

      {/* ── ARYA'S QUESTION ── */}
      <section className="review-section">
        <p className="review-section-label">ARYA'S QUESTION FOR YOU</p>
        <p className="arya-question">
          "{review.aryasQuestion}"
        </p>
        <textarea
          className="question-response"
          placeholder="Sit with this. Write what comes..."
          value={userResponse}
          onChange={e => setUserResponse(e.target.value)}
        />
      </section>

      <div className="review-section-divider" />

      {/* ── INTENTION NEXT WEEK ── */}
      <section className="review-section">
        <p className="review-section-label">
          YOUR INTENTION NEXT WEEK
        </p>
        <div className="intention-grid">
          {review.intentionOptions.map((option, i) => (
            <button
              key={i}
              className={`intention-chip ${
                selectedIntention === option ? 'selected' : ''
              }`}
              onClick={() => setSelectedIntention(option)}
            >
              {option}
            </button>
          ))}
        </div>
        <input
          className="intention-custom"
          placeholder="Or write your own..."
          value={customIntention}
          onChange={e => setCustomIntention(e.target.value)}
        />
        <button
          className="intention-set-btn"
          onClick={() => saveIntention(
            selectedIntention || customIntention
          )}
          disabled={!selectedIntention && !customIntention}
        >
          Set my intention for next week
        </button>
      </section>

      {/* ── FOOTER ── */}
      <p className="review-footer">
        ARYA · Your personal thinking partner
      </p>

    </div>
  );
}
```
