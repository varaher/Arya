## Solution 1 — Auto Health Check (Best Option)

```typescript
// server/arya/sarvam-health-check.ts
// Run this every day at midnight

async function checkSarvamSpeakers() {
  const testText = "hello";
  const speakers = [
    'priya', 'anushka', 'kavitha',
    'simran', 'manisha', 'vidya'
  ];

  const working = [];
  const broken = [];

  for (const speaker of speakers) {
    try {
      const response = await fetch(
        'https://api.sarvam.ai/text-to-speech',
        {
          method: 'POST',
          headers: {
            'API-Subscription-Key': process.env.SARVAM_API_KEY!,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            inputs: [testText],
            target_language_code: 'hi-IN',
            speaker: speaker,
            model: 'bulbul:v2',
          })
        }
      );

      if (response.ok) {
        working.push(speaker);
      } else {
        const err = await response.json();
        broken.push({ speaker, error: err });
      }
    } catch (e) {
      broken.push({ speaker, error: e });
    }
  }

  console.log('[Sarvam Health] Working:', working);
  console.log('[Sarvam Health] Broken:', broken);

  // If any current speaker is broken → alert
  const currentSpeakers = [
    'priya', 'anushka', 'kavitha', 'simrin'
  ];

  const brokenCurrent = broken.filter(b =>
    currentSpeakers.includes(b.speaker)
  );

  if (brokenCurrent.length > 0) {
    console.error(
      '[ALERT] Sarvam speakers broken:',
      brokenCurrent
    );
    // Send notification to admin
  }

  return { working, broken };
}
```

## Solution 2 — Dynamic Speaker Detection

```typescript
// Instead of hardcoding speaker names
// Detect which speakers work at runtime

// server/arya/sarvam-service.ts

let workingSpeakers: string[] = [];
let lastChecked: Date | null = null;

async function getWorkingSpeakers():
  Promise<string[]> {

  // Cache for 24 hours
  const now = new Date();
  if (
    lastChecked &&
    now.getTime() - lastChecked.getTime()
    < 24 * 60 * 60 * 1000 &&
    workingSpeakers.length > 0
  ) {
    return workingSpeakers;
  }

  // Test all known speakers
  const allSpeakers = [
    'priya', 'anushka', 'kavitha',
    'simran', 'manisha', 'vidya',
    'arjun', 'meera', 'ananya'
  ];

  const working = [];
  for (const speaker of allSpeakers) {
    const ok = await testSpeaker(speaker);
    if (ok) working.push(speaker);
  }

  workingSpeakers = working;
  lastChecked = now;

  console.log('[Sarvam] Working speakers:', working);
  return working;
}

// Use working speakers dynamically:
export function getSpeakerForLanguage(
  langCode: string
): string {
  // Language preference map
  const preference: Record<string, string[]> = {
    'ml-IN': ['anushka', 'vidya', 'priya'],
    'hi-IN': ['priya', 'manisha', 'vidya'],
    'ta-IN': ['kavitha', 'priya', 'vidya'],
    'te-IN': ['anushka', 'priya', 'vidya'],
    'kn-IN': ['anushka', 'priya', 'vidya'],
    'pa-IN': ['simran', 'priya', 'vidya'],
    'en-IN': ['priya', 'vidya', 'manisha'],
  };

  const preferred = preference[langCode] || ['priya'];

  // Return first working speaker from preference list
  for (const speaker of preferred) {
    if (workingSpeakers.includes(speaker)) {
      return speaker;
    }
  }

  // Last resort — first working speaker
  return workingSpeakers[0] || 'priya';
}
```

## Solution 4 — Read Available Speakers From Error

```typescript
// The smartest fix — when Sarvam says
// "Speaker X not found. Available: A, B, C"
// Automatically use A and remember it

async function handleTTSError(
  errorBody: any,
  text: string,
  languageCode: string
): Promise<TTSResult | null> {

  const message = errorBody?.error?.message || '';

  // Sarvam tells us what's available
  if (message.includes('Available speakers are:')) {
    const available = message
      .split('Available speakers are: ')[1]
      ?.split(/[,\n]/)[0]
      ?.trim();

    if (available) {
      console.log(
        `[Auto-fix] Sarvam retired speaker.
         Switching to: ${available}`
      );

      // Update the speaker map
      updateSpeakerMap(languageCode, available);

      // Retry immediately with new speaker
      return await sarvamTextToSpeech(
        text,
        languageCode,
        available
      );
    }
  }

  return null;
}
```

This is the most elegant solution — Sarvam's own error message tells us what speakers are available. We read the error and automatically switch. No manual fix needed ever again.
