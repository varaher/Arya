## The Fix — Exact Code

### Fix 1 — Save profile to DB (routes.ts)

```typescript
// Add this route if it doesn't exist:
app.post('/api/user/kaal-profile', requireAuth, async (req, res) => {
  const userId = req.user.id;
  const {
    birthDate,
    birthTime,
    birthCity,
    starSign,      // Western
    moonSign,      // Vedic Rashi
    setupComplete,
  } = req.body;

  // Upsert — insert or update
  await db
    .insert(arya_kaal_profiles)
    .values({
      userId,
      birthDate,
      birthTime,
      birthCity,
      starSign,
      moonSign,
      setupComplete: true,
      updatedAt: new Date(),
    })
    .onConflictDoUpdate({
      target: arya_kaal_profiles.userId,
      set: {
        birthDate,
        birthTime,
        birthCity,
        starSign,
        moonSign,
        setupComplete: true,
        updatedAt: new Date(),
      }
    });

  res.json({ success: true });
});

// Add GET route:
app.get('/api/user/kaal-profile', requireAuth, async (req, res) => {
  const userId = req.user.id;

  const profile = await db
    .select()
    .from(arya_kaal_profiles)
    .where(eq(arya_kaal_profiles.userId, userId))
    .limit(1);

  res.json(profile[0] || null);
});
```

### Fix 2 — Check profile on KAAL open (VedicLensPage.tsx)

```typescript
// At the top of VedicLensPage component:

const [profileLoaded, setProfileLoaded] = useState(false);
const [hasProfile, setHasProfile] = useState(false);
const [kaalProfile, setKaalProfile] = useState(null);

useEffect(() => {
  // Check if profile exists when component mounts
  fetch('/api/user/kaal-profile')
    .then(r => r.json())
    .then(profile => {
      if (profile && profile.setupComplete) {
        setHasProfile(true);     // ← skip setup
        setKaalProfile(profile); // ← use saved data
      } else {
        setHasProfile(false);    // ← show setup
      }
      setProfileLoaded(true);
    })
    .catch(() => {
      setHasProfile(false);
      setProfileLoaded(true);
    });
}, []); // runs once on mount

// In JSX:
if (!profileLoaded) {
  return <LoadingSpinner />; // don't flash setup form
}

if (!hasProfile) {
  return <KaalSetupForm onComplete={handleSetupComplete} />;
}

return <KaalDashboard profile={kaalProfile} />;
```

### Fix 3 — Save on form completion

```typescript
// In the KAAL setup form — onComplete handler:

async function handleSetupComplete(formData) {
  try {
    // Save to server
    const res = await fetch('/api/user/kaal-profile', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        ...formData,
        setupComplete: true,
      }),
    });

    if (!res.ok) throw new Error('Save failed');

    // Update local state
    setKaalProfile(formData);
    setHasProfile(true); // ← now shows dashboard

  } catch (err) {
    console.error('KAAL profile save failed:', err);
    // Show error to user
  }
}
```

### Fix 4 — DB table (if not exists)

```sql
CREATE TABLE IF NOT EXISTS arya_kaal_profiles (
  id              SERIAL PRIMARY KEY,
  user_id         INTEGER UNIQUE NOT NULL REFERENCES users(id),
  birth_date      TEXT,
  birth_time      TEXT,
  birth_city      TEXT,
  star_sign       TEXT,    -- Western: Aries, Taurus etc
  moon_sign       TEXT,    -- Vedic Rashi: Mesh, Vrishabh etc
  setup_complete  BOOLEAN DEFAULT FALSE,
  created_at      TIMESTAMP DEFAULT NOW(),
  updated_at      TIMESTAMP DEFAULT NOW()
);
```
