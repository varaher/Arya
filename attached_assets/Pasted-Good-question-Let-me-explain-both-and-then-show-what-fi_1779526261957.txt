## The Fix — Gemini-Style Auto Silence Detection

This is the most important missing piece. Instead of a stop button, ARYA detects when you stop talking:

```typescript
// client/src/hooks/useVoiceRecorder.ts

export function useVoiceRecorder(onComplete: (blob: Blob) => void) {
  const [isListening, setIsListening] = useState(false);
  const silenceTimer = useRef<NodeJS.Timeout | null>(null);
  const analyserRef = useRef<AnalyserNode | null>(null);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const chunksRef = useRef<Blob[]>([]);

  async function startListening() {
    const stream = await navigator.mediaDevices.getUserMedia({
      audio: {
        echoCancellation: true,
        noiseSuppression: true,
        sampleRate: 16000,
      }
    });

    // Set up audio analyser for silence detection
    const audioCtx = new AudioContext();
    const analyser = audioCtx.createAnalyser();
    analyser.fftSize = 256;
    const source = audioCtx.createMediaStreamSource(stream);
    source.connect(analyser);
    analyserRef.current = analyser;

    // Start recorder
    const recorder = new MediaRecorder(stream);
    recorder.ondataavailable = (e) => {
      if (e.data.size > 0) chunksRef.current.push(e.data);
    };
    recorder.onstop = async () => {
      stream.getTracks().forEach(t => t.stop());
      setIsListening(false);
      const blob = new Blob(chunksRef.current, {
        type: recorder.mimeType
      });
      chunksRef.current = [];
      onComplete(blob);  // send to ARYA
    };

    recorder.start(100);
    mediaRecorderRef.current = recorder;
    setIsListening(true);

    // Start silence detection loop
    detectSilence();
  }

  function detectSilence() {
    const analyser = analyserRef.current;
    if (!analyser) return;

    const data = new Uint8Array(analyser.frequencyBinCount);

    function check() {
      if (!mediaRecorderRef.current) return;

      analyser.getByteFrequencyData(data);
      const volume = data.reduce((a, b) => a + b) / data.length;

      if (volume < 10) {
        // Silence detected — start 1.5 second timer
        if (!silenceTimer.current) {
          silenceTimer.current = setTimeout(() => {
            // Stop recording after 1.5s of silence
            mediaRecorderRef.current?.stop();
            silenceTimer.current = null;
          }, 1500);
        }
      } else {
        // Sound detected — cancel silence timer
        if (silenceTimer.current) {
          clearTimeout(silenceTimer.current);
          silenceTimer.current = null;
        }
      }

      requestAnimationFrame(check);
    }

    check();
  }

  function stopListening() {
    mediaRecorderRef.current?.stop();
    if (silenceTimer.current) {
      clearTimeout(silenceTimer.current);
    }
  }

  return { isListening, startListening, stopListening };
}
```
