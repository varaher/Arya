## The Two Fixes — Exact Code

### Fix 1 — Always Auto-detect in sarvam-service.ts

Find the `transcribe()` function and change one line:

```typescript
// BEFORE:
body: JSON.stringify({
  model: 'saarika:v2.5',
  language_code: languageCode,  // fixed language
  audio: audioBase64,
})

// AFTER:
body: JSON.stringify({
  model: 'saarika:v2.5',
  language_code: 'unknown',     // always auto-detect
  audio: audioBase64,
})

// Also return the detected language:
return {
  transcript: data.transcript,
  detectedLanguage: data.language_code || languageCode,
}
```

### Fix 2 — International TTS in routes.ts

Find the voice response TTS section:

```typescript
// BEFORE:
// Only Sarvam or OpenAI nova

// AFTER:
const INDIAN_LANGS = [
  'hi-IN', 'ml-IN', 'ta-IN', 'te-IN',
  'kn-IN', 'bn-IN', 'mr-IN', 'gu-IN',
  'pa-IN', 'od-IN', 'en-IN'
];

const INTERNATIONAL_VOICES: Record<string, string> = {
  'fr': 'alloy',
  'de': 'nova',
  'es': 'shimmer',
  'ar': 'echo',
  'ja': 'fable',
  'zh': 'onyx',
  'ko': 'alloy',
  'ru': 'nova',
  'pt': 'shimmer',
  'tr': 'echo',
  'id': 'alloy',
  'sw': 'nova',
  'he': 'fable',
};

// TTS routing:
let audioBase64: string | null = null;

if (INDIAN_LANGS.includes(detectedLanguage)) {
  // Indian language → Sarvam
  audioBase64 = await sarvamService.textToSpeech(
    responseText,
    detectedLanguage
  );
} else {
  // International → OpenAI TTS
  const langCode = detectedLanguage.split('-')[0];
  const voice = INTERNATIONAL_VOICES[langCode] || 'nova';
  
  const speech = await openai.audio.speech.create({
    model: 'tts-1',
    voice,
    input: responseText,
  });
  
  const buffer = Buffer.from(await speech.arrayBuffer());
  audioBase64 = buffer.toString('base64');
}
```
