// ARYA Reminder Sound Utility
// Indian-inspired chime using Web Audio API
// No audio files needed — pure code music

export type SoundType = 
  | 'reminder'      // Soft temple bell
  | 'goal'          // Rising sitar-like tone  
  | 'morning'       // Peaceful raga opener
  | 'success'       // Celebratory ascending notes
  | 'gentle'        // Minimal single tone

let audioCtx: AudioContext | null = null;

function getAudioContext(): AudioContext {
  if (!audioCtx || audioCtx.state === 'closed') {
    audioCtx = new (window.AudioContext || (window as any).webkitAudioContext)();
  }
  if (audioCtx.state === 'suspended') {
    audioCtx.resume();
  }
  return audioCtx;
}

// ── Core tone builder ─────────────────────────────────────────────────────────
function playTone(
  ctx: AudioContext,
  frequency: number,
  startTime: number,
  duration: number,
  volume: number = 0.3,
  type: OscillatorType = 'sine',
  decayStyle: 'bell' | 'pluck' | 'sustain' = 'bell'
) {
  const oscillator = ctx.createOscillator();
  const gainNode = ctx.createGain();
  const compressor = ctx.createDynamicsCompressor();

  oscillator.connect(gainNode);
  gainNode.connect(compressor);
  compressor.connect(ctx.destination);

  oscillator.type = type;
  oscillator.frequency.setValueAtTime(frequency, startTime);

  // Decay styles
  gainNode.gain.setValueAtTime(0, startTime);
  gainNode.gain.linearRampToValueAtTime(volume, startTime + 0.01); // quick attack

  if (decayStyle === 'bell') {
    // Bell: fast attack, long exponential decay
    gainNode.gain.exponentialRampToValueAtTime(0.001, startTime + duration);
  } else if (decayStyle === 'pluck') {
    // Pluck: fast attack, medium decay
    gainNode.gain.exponentialRampToValueAtTime(0.001, startTime + duration * 0.6);
  } else {
    // Sustain: hold then fade
    gainNode.gain.setValueAtTime(volume, startTime + duration * 0.7);
    gainNode.gain.linearRampToValueAtTime(0, startTime + duration);
  }

  oscillator.start(startTime);
  oscillator.stop(startTime + duration + 0.1);
}

// Add harmonics for richer bell-like tone
function playRichTone(
  ctx: AudioContext,
  frequency: number,
  startTime: number,
  duration: number,
  volume: number = 0.25
) {
  // Fundamental
  playTone(ctx, frequency, startTime, duration, volume, 'sine', 'bell');
  // 2nd harmonic (octave) — quieter
  playTone(ctx, frequency * 2, startTime, duration * 0.7, volume * 0.3, 'sine', 'bell');
  // 3rd harmonic — very quiet
  playTone(ctx, frequency * 3, startTime, duration * 0.4, volume * 0.1, 'sine', 'bell');
}

// ── SOUND 1: Temple Bell (default reminder) ───────────────────────────────────
// Sa-Pa-Sa — the most calming Indian interval
export function playTempleBell() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Sa (C5 = 523Hz), Pa (G5 = 784Hz), Sa (C6 = 1047Hz)
  // Three ascending bell strikes with reverb-like decay
  playRichTone(ctx, 523.25, now,        2.5, 0.28);
  playRichTone(ctx, 784.00, now + 0.5,  2.0, 0.22);
  playRichTone(ctx, 1046.5, now + 0.9,  2.5, 0.18);

  // Soft resonance underneath
  playTone(ctx, 261.63, now, 3.5, 0.08, 'sine', 'bell'); // C4 drone
}

// ── SOUND 2: Sitar Pluck (goal achieved / added) ──────────────────────────────
// Rising Sa-Re-Ga-Ma — hopeful ascending
export function playSitarRise() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Indian scale notes: Sa Re Ga Ma Pa
  const notes = [261.63, 293.66, 329.63, 349.23, 392.00];
  const delays = [0, 0.15, 0.28, 0.40, 0.52];

  notes.forEach((freq, i) => {
    // Main tone
    playTone(ctx, freq, now + delays[i], 1.2, 0.25, 'sawtooth', 'pluck');
    // Shimmer harmonic (sitar-like)
    playTone(ctx, freq * 2.01, now + delays[i] + 0.01, 0.8, 0.08, 'sine', 'pluck');
  });

  // Final bell resolution
  playRichTone(ctx, 523.25, now + 0.7, 2.0, 0.2);
}

// ── SOUND 3: Morning Raga (morning reminder / Sunday review) ──────────────────
// Bhairav raga opening phrase — peaceful, grounding
export function playMorningRaga() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Bhairav: Sa Re(komal) Ga Ma Pa Dha(komal) Ni Sa
  // Frequencies for C-based Bhairav
  const raga = [
    { freq: 261.63, delay: 0,    dur: 0.6 },  // Sa
    { freq: 277.18, delay: 0.5,  dur: 0.4 },  // Re komal (C#)
    { freq: 329.63, delay: 0.85, dur: 0.5 },  // Ga (E)
    { freq: 349.23, delay: 1.3,  dur: 0.6 },  // Ma (F)
    { freq: 392.00, delay: 1.85, dur: 0.5 },  // Pa (G)
    { freq: 415.30, delay: 2.3,  dur: 0.4 },  // Dha komal (G#)
    { freq: 493.88, delay: 2.65, dur: 0.5 },  // Ni (B)
    { freq: 523.25, delay: 3.1,  dur: 1.5 },  // Sa (C5) — hold
  ];

  raga.forEach(({ freq, delay, dur }) => {
    playTone(ctx, freq, now + delay, dur, 0.2, 'sine', 'sustain');
    // Tanpura drone underneath
    playTone(ctx, 65.41, now + delay, dur, 0.04, 'sine', 'sustain'); // C2 drone
  });

  // Closing bell
  playRichTone(ctx, 523.25, now + 4.0, 3.0, 0.15);
}

// ── SOUND 4: Success Chime (task done / goal complete) ────────────────────────
// Quick joyful ascending + resolve
export function playSuccessChime() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Quick ascending: C E G C (major arpeggio — universal joy)
  const chord = [523.25, 659.25, 783.99, 1046.5];
  chord.forEach((freq, i) => {
    playRichTone(ctx, freq, now + i * 0.12, 1.5, 0.22);
  });

  // Resolution shimmer
  playTone(ctx, 1318.5, now + 0.5, 1.0, 0.1, 'sine', 'bell'); // E6
}

// ── SOUND 5: Gentle Ping (soft notification) ──────────────────────────────────
// Single pure tone — minimal, not disruptive
export function playGentlePing() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  playRichTone(ctx, 659.25, now, 2.0, 0.2); // E5 — warm, not harsh
}

// ── SOUND 6: ARYA Wake (app open / greeting) ──────────────────────────────────
// Two-note call — like ARYA saying hello
export function playAryaWake() {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Perfect fifth — G then C (resolving home)
  playRichTone(ctx, 392.00, now,        1.5, 0.2);  // G4
  playRichTone(ctx, 523.25, now + 0.35, 2.5, 0.22); // C5
  
  // Soft sparkle
  playTone(ctx, 1046.5, now + 0.6, 1.5, 0.06, 'sine', 'bell'); // C6
}

// ── Main dispatcher ───────────────────────────────────────────────────────────
export function playARYASound(type: SoundType = 'reminder') {
  try {
    switch (type) {
      case 'reminder': playTempleBell();   break;
      case 'goal':     playSitarRise();    break;
      case 'morning':  playMorningRaga();  break;
      case 'success':  playSuccessChime(); break;
      case 'gentle':   playGentlePing();   break;
      default:         playTempleBell();
    }
  } catch (err) {
    console.warn('ARYA sound blocked by browser autoplay policy:', err);
    // Silently fail — notification still shows
  }
}

//
// Import anywhere in your React app:
//   import { playARYASound } from '@/utils/reminderSound';
//
// When reminder fires:
//   playARYASound('reminder');   → Temple bell (Sa-Pa-Sa)
//
// When goal is added:
//   playARYASound('goal');       → Sitar rise
//
// When goal is completed:
//   playARYASound('success');    → Success chime
//
// Morning reminder:
//   playARYASound('morning');    → Bhairav raga phrase
//
// Soft notification:
//   playARYASound('gentle');     → Single ping
//
// App greeting:
//   playAryaWake();              → Two-note ARYA hello
//
// NOTE: First call must happen after user interaction (tap/click)
// due to browser autoplay policy. The AudioContext resumes automatically.
