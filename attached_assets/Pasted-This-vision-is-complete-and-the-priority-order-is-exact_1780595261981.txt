// After goal saved — check if vague:
function isVagueGoal(title: string): boolean {
  const vagueWords = /^(get|be|become|stay|feel|have|do|make|learn|improve|start|stop|try)\s/i;
  const specificMarkers = /\d|by\s|every\s|per\s|times?\s|km|kg|min|hour|book|page/i;
  return vagueWords.test(title) && !specificMarkers.test(title);
}

if (isVagueGoal(goal.title)) {
  // Store flag — ARYA asks specificity question next message
  await db.update(arya_conversations)
    .set({ pendingSpecificityGoalId: goal.id })

}

### Step 3 — Evening Check-in Notification (Notification Flow)

In `reminder-scheduler.ts`, add an evening sweep at 8 PM:

```typescript
// Runs daily at 8 PM
async function sendEveningGoalCheckins() {
  const users = await getUsersWithActiveGoals();

  for (const user of users) {
    const topGoal = user.goals
      .filter(g => g.progress === 0)
      .sort((a, b) => 
        new Date(a.createdAt).getTime() - 
        new Date(b.createdAt).getTime()
      )[0];

    if (!topGoal) continue;

    // Don't send if already checked in today
    if (topGoal.lastCheckedAt) {
      const hoursAgo = (Date.now() - new Date(topGoal.lastCheckedAt).getTime()) 
                       / (1000 * 60 * 60);
      if (hoursAgo < 20) continue;
    }

    await sendPushNotification(user.id, {
      title: `🎯 ${topGoal.title}`,
      body: `Did you work on this today?`,
      type: 'goal_checkin',
      goalId: topGoal.id,
      actions: [
        { action: 'yes',  title: '✅ Yes' },
        { action: 'no',   title: '❌ Not yet' },
        { action: 'skip', title: '↩ Skip today' }
      ]
    });
  }
}
```
