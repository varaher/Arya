# ARYA — 45-Day Trial System
## Complete Developer Specification

---

## Overview

Every new user gets 45 days of full Pro access.
No credit card required at signup.
On Day 45 — they choose a plan or move to Free.
Beta users (existing 312) get special founding member treatment.

---

## 1. Database Changes

### 1A — Add trial columns to users table

```sql
ALTER TABLE users
  ADD COLUMN IF NOT EXISTS trial_started_at   TIMESTAMP,
  ADD COLUMN IF NOT EXISTS trial_ends_at      TIMESTAMP,
  ADD COLUMN IF NOT EXISTS trial_status       TEXT DEFAULT 'active',
  -- 'active' | 'expired' | 'converted' | 'founding_member'
  ADD COLUMN IF NOT EXISTS is_founding_member BOOLEAN DEFAULT FALSE,
  ADD COLUMN IF NOT EXISTS founding_price     INTEGER;
  -- stores ₹149 in paise (14900) for founding members
```

### 1B — Create trial notifications tracking table

```sql
CREATE TABLE IF NOT EXISTS arya_trial_notifications (
  id              SERIAL PRIMARY KEY,
  user_id         INTEGER NOT NULL REFERENCES users(id),
  day_number      INTEGER NOT NULL,
  notification_type TEXT NOT NULL,
  -- 'discovery' | 'nudge' | 'upgrade_soft' | 'upgrade_final'
  title           TEXT NOT NULL,
  body            TEXT NOT NULL,
  scheduled_at    TIMESTAMP NOT NULL,
  sent_at         TIMESTAMP,
  opened_at       TIMESTAMP,
  converted       BOOLEAN DEFAULT FALSE,
  created_at      TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_trial_notif_user ON arya_trial_notifications(user_id);
CREATE INDEX idx_trial_notif_scheduled ON arya_trial_notifications(scheduled_at);
```

```typescript
      title: n.title,
      body: n.body,
```

## 3. Dynamic Day 40 Message

The Day 40 notification must be personalised. Generate it at send time:

```typescript
// server/arya/trial-notifications.ts

async function buildDay40Body(userId: number): Promise<string> {
  // Count user's activity
  const [convCount] = await db
    .select({ count: count() })
    .from(arya_conversations)
    .where(eq(arya_conversations.userId, userId));

  const [goalCount] = await db
    .select({ count: count() })
    .from(arya_goals)
    .where(eq(arya_goals.userId, userId));

  const [completedCount] = await db
    .select({ count: count() })
    .from(arya_goals)
    .where(
      and(
        eq(arya_goals.userId, userId),
        eq(arya_goals.status, 'completed')
      )
    );

  const [noteCount] = await db
    .select({ count: count() })
    .from(arya_voice_notes)
    .where(eq(arya_voice_notes.userId, userId));

  const conversations = convCount?.count || 0;
  const goals = goalCount?.count || 0;
  const completed = completedCount?.count || 0;
  const notes = noteCount?.count || 0;

  return `You've been with ARYA for 40 days.

${conversations} conversations.
${goals} goals set${completed > 0 ? `, ${completed} completed` : ''}.
${notes > 0 ? `${notes} voice notes recorded.` : ''}

ARYA knows you now.
5 more days of full access — then choose how you continue.

Core ₹249/month — everything you've been using, continues.`;
}
```

## 4. Notification Scheduler — Runs Every 10 Minutes

```typescript
// Add to server/arya/reminder-scheduler.ts
// Or create server/arya/trial-scheduler.ts

export async function processTrialNotifications(): Promise<void> {
  const now = new Date();
  const tenMinutesAgo = new Date(now.getTime() - 10 * 60 * 1000);

  // Find all unsent notifications due in the last 10 minutes
  const due = await db
    .select()
    .from(arya_trial_notifications)
    .where(
      and(
        isNull(arya_trial_notifications.sentAt),
        lte(arya_trial_notifications.scheduledAt, now),
        gte(arya_trial_notifications.scheduledAt, tenMinutesAgo)
      )
    )
    .limit(50);

  for (const notification of due) {
    try {
      let body = notification.body;

      // Day 40: build dynamic body
      if (notification.dayNumber === 40) {
        body = await buildDay40Body(notification.userId);
      }

      // Send push notification
      await sendPushNotification(notification.userId, {
        title: notification.title,
        body,
        type: 'trial',
        dayNumber: notification.dayNumber,
        data: {
          url: getNotificationUrl(notification.dayNumber),
          trialDay: notification.dayNumber,
        },
        // Upgrade notifications: require interaction
        requireInteraction: [40, 43, 45].includes(notification.dayNumber),
        actions: [40, 43, 45].includes(notification.dayNumber)
          ? [
              { action: 'upgrade', title: '🚀 See plans' },
              { action: 'dismiss', title: 'Later' },
            ]
          : undefined,
      });

      // Mark as sent
      await db
        .update(arya_trial_notifications)
        .set({ sentAt: new Date() })
        .where(eq(arya_trial_notifications.id, notification.id));

    } catch (err) {
      console.error(
        `Trial notification failed: user=${notification.userId} day=${notification.dayNumber}`,
        err
      );
    }
  }
}

// URL to open when notification is tapped
```

## 5. Trial Status Check — Middleware

```typescript
// server/middleware/trial-check.ts

export async function getEffectivePlan(userId: number): Promise<string> {
  const user = await db
    .select()
    .from(users)
    .where(eq(users.id, userId))
    .limit(1);

  if (!user[0]) return 'free';

  const u = user[0];

  // Paid subscriber — return their plan
  if (u.subscriptionPlan && u.subscriptionStatus === 'active') {
    return u.subscriptionPlan;
  }

  // Active trial — return 'pro' (full access)
  if (u.trialStatus === 'active' && u.trialEndsAt) {
    if (new Date() < new Date(u.trialEndsAt)) {
      return 'pro'; // trial = full pro access
    } else {
      // Trial expired — update status
      await db
        .update(users)
        .set({ trialStatus: 'expired' })
        .where(eq(users.id, userId));
      return 'free';
    }
  }

  return 'free';
}
```

## 6. Trial Status API — Frontend

```typescript
// Add to routes.ts:

app.get('/api/user/trial-status', requireAuth, async (req, res) => {
  const userId = req.user.id;
  const user = await db
    .select()
    .from(users)
    .where(eq(users.id, userId))
    .limit(1);

  const u = user[0];
  const now = new Date();
  const trialEnds = u.trialEndsAt ? new Date(u.trialEndsAt) : null;
  const daysLeft = trialEnds
    ? Math.max(0, Math.ceil(
        (trialEnds.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
      ))
    : 0;

  const effectivePlan = await getEffectivePlan(userId);

  res.json({
    trialStatus: u.trialStatus,           // 'active' | 'expired' | 'converted'
    trialStartedAt: u.trialStartedAt,
    trialEndsAt: u.trialEndsAt,
    daysLeft,
    effectivePlan,                         // 'free' | 'pro' | 'elite' etc
    isFoundingMember: u.isFoundingMember,
    foundingPrice: u.foundingPrice,
    isPaidSubscriber: u.subscriptionStatus === 'active',
  });
});
```

```typescript
export async function markFoundingMembers(): Promise<void> {
```
