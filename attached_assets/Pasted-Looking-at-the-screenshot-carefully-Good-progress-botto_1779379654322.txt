## Issue 1 — No Stop Button When Recording

This is the most critical UX bug. User taps mic → recording starts → no way to stop it.

```typescript
// client/src/components/VoiceRecordButton.tsx
// Complete replacement of mic button UI

import { useState, useEffect, useRef } from 'react';

interface VoiceRecordButtonProps {
  onRecordingComplete: (blob: Blob) => void;
  isProcessing: boolean;
}

export function VoiceRecordButton({
  onRecordingComplete,
  isProcessing
}: VoiceRecordButtonProps) {
  const [state, setState] = useState<
    'idle' | 'recording' | 'processing'
  >('idle');
  const [duration, setDuration] = useState(0);
  const [amplitude, setAmplitude] = useState(0);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const chunksRef = useRef<Blob[]>([]);
  const timerRef = useRef<NodeJS.Timeout | null>(null);
  const analyserRef = useRef<AnalyserNode | null>(null);
  const animFrameRef = useRef<number | null>(null);

  // Timer while recording
  useEffect(() => {
    if (state === 'recording') {
      timerRef.current = setInterval(() => {
        setDuration(d => d + 1);
      }, 1000);
    } else {
      if (timerRef.current) clearInterval(timerRef.current);
      setDuration(0);
    }
    return () => {
      if (timerRef.current) clearInterval(timerRef.current);
    };
  }, [state]);

  // Amplitude animation while recording
  function startAmplitudeTracking(stream: MediaStream) {
    const audioCtx = new AudioContext();
    const analyser = audioCtx.createAnalyser();
    analyser.fftSize = 256;
    const source = audioCtx.createMediaStreamSource(stream);
    source.connect(analyser);
    analyserRef.current = analyser;

    const data = new Uint8Array(analyser.frequencyBinCount);

    function tick() {
      analyser.getByteFrequencyData(data);
      const avg = data.reduce((a, b) => a + b, 0) / data.length;
      setAmplitude(avg);
      animFrameRef.current = requestAnimationFrame(tick);
    }
    tick();
  }

  function stopAmplitudeTracking() {
    if (animFrameRef.current) {
      cancelAnimationFrame(animFrameRef.current);
    }
    setAmplitude(0);
  }

  async function startRecording() {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          sampleRate: 16000,
        }
      });

      startAmplitudeTracking(stream);
      chunksRef.current = [];

      const mimeType = getBestFormat();
      const recorder = mimeType
        ? new MediaRecorder(stream, { mimeType })
        : new MediaRecorder(stream);

      recorder.ondataavailable = (e) => {
        if (e.data.size > 0) chunksRef.current.push(e.data);
      };

      recorder.onstop = async () => {
        stopAmplitudeTracking();
        stream.getTracks().forEach(t => t.stop());
        setState('processing');

        const rawBlob = new Blob(chunksRef.current, {
          type: recorder.mimeType
        });

        // Convert to WAV
        const { convertToWav } = await import('../utils/audioConverter');
        const wavBlob = await convertToWav(rawBlob);
        onRecordingComplete(wavBlob);
      };

      recorder.start(100); // collect chunks every 100ms
      mediaRecorderRef.current = recorder;
      setState('recording');

    } catch (err: any) {
      if (err.name === 'NotAllowedError') {
        alert('Microphone access denied. Please allow mic in browser settings.');
      } else {
        alert('Could not start recording. Please try again.');
      }
      console.error('[voice] Start error:', err);
    }
  }

  function stopRecording() {
    if (mediaRecorderRef.current?.state === 'recording') {
      mediaRecorderRef.current.stop();
    }
  }

  function formatDuration(secs: number): string {
    const m = Math.floor(secs / 60);
    const s = secs % 60;
    return `${m}:${s.toString().padStart(2, '0')}`;
  }

  function getBestFormat(): string {
    const formats = [
      'audio/webm;codecs=opus',
      'audio/webm',
      'audio/ogg;codecs=opus',
      'audio/mp4',
    ];
    return formats.find(f => MediaRecorder.isTypeSupported(f)) || '';
  }

  // ── IDLE STATE — just mic button ──
  if (state === 'idle') {
    return (
      <button
        className="voice-mic-btn"
        onClick={startRecording}
        title="Tap to record"
      >
        🎙️
      </button>
    );
  }

  // ── RECORDING STATE — waveform + stop button ──
  if (state === 'recording') {
    const scale = 1 + (amplitude / 128) * 0.4;

    return (
      <div className="voice-recording-bar">

        {/* Pulsing mic */}
        <div
          className="vrb-mic-pulse"
          style={{ transform: `scale(${scale})` }}
        >
          🎙️
        </div>

        {/* Timer */}
        <span className="vrb-timer">
          {formatDuration(duration)}
        </span>

        {/* Waveform animation */}
        <div className="vrb-wave">
          {[...Array(5)].map((_, i) => (
            <div
              key={i}
              className="vrb-bar"
              style={{
                height: `${8 + (amplitude / 255) * 20 *
                  Math.sin(Date.now() / 200 + i)}px`,
                animationDelay: `${i * 0.1}s`
              }}
            />
          ))}
        </div>

        {/* STOP BUTTON — prominent, can't miss it */}
        <button
          className="vrb-stop-btn"
          onClick={stopRecording}
          title="Stop recording"
        >
          ⏹ Stop
        </button>

      </div>
    );
  }

  // ── PROCESSING STATE ──
  if (state === 'processing') {
    return (
      <div className="voice-processing">
        <div className="vp-spinner" />
        <span>Transcribing...</span>
      </div>
    );
  }

  return null;
}
```

## Issue 3 — Transcription Speed

The delay happens in two places:

```
User stops recording
        ↓
WAV conversion    → 200-500ms  (client side)
        ↓
Upload to server  → 500ms-2s   (depends on file size + network)
        ↓
Sarvam STT API    → 1-3s       (the main delay)
        ↓
Response back     → 200ms
        ↓
Total: 2-6 seconds
```

**Three things to speed this up:**

```typescript
// Fix 1 — Start processing BEFORE user fully stops
// Begin sending audio while still recording (streaming)

recorder.ondataavailable = async (e) => {
  if (e.data.size > 0) {
    chunksRef.current.push(e.data);
  }
};

// Change chunk interval from 100ms to 250ms
// Larger chunks = fewer network round trips
recorder.start(250);


// Fix 2 — Reduce WAV sample rate for faster upload
// 16000 Hz is sufficient for speech recognition
// (already set, confirm it's being used)

const audioContext = new AudioCtx({ sampleRate: 16000 });


// Fix 3 — Show "Transcribing..." immediately
// Don't wait for server to respond before showing feedback

recorder.onstop = async () => {
  setState('processing'); // ← show spinner IMMEDIATELY
  stopAmplitudeTracking();
  stream.getTracks().forEach(t => t.stop());

  const rawBlob = new Blob(chunksRef.current, {
    type: recorder.mimeType || 'audio/webm'
  });

  // Convert and send
  const wavBlob = await convertToWav(rawBlob);
  onRecordingComplete(wavBlob);
};
```
