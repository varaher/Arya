// ARYA Document Extractor — Complete Implementation
//
// What this does:
// After a document is uploaded, ARYA reads it and pulls out:
//   → Key points (summary)
//   → Tasks / things to do
//   → Dates and deadlines
//   → Exam questions (if study content)
//   → Important terms
//
// Then connects everything to Goals, Reminders, Notes, and Chat.
//
// File: server/arya/document-processor.ts
// ═══════════════════════════════════════════════════════════════════════════


// ─────────────────────────────────────────────────────────────────────────────
// TYPES
// ─────────────────────────────────────────────────────────────────────────────

export interface ExtractedDate {
  label: string;        // "Night duty" / "NABH audit" / "Chapter test"
  dateText: string;     // "June 18, 8 PM" — exactly as it appears in document
  isoDate?: string;     // "2026-06-18T20:00:00" — parsed if possible
}

export interface DocumentExtracts {
  summary: string[];          // 4-6 bullet points
  tasks: string[];            // Action items found
  dates: ExtractedDate[];     // Dates and deadlines
  examQuestions: string[];    // Generated from study content
  terms: string[];            // "Term — definition" strings
  isStudyContent: boolean;    // True if textbook/notes/exam prep
  hasDates: boolean;          // True if any dates found
  hasTasks: boolean;          // True if any tasks found
}


// ─────────────────────────────────────────────────────────────────────────────
// THE EXTRACTION PROMPT
// What we ask GPT-4o-mini to do with the document
// ─────────────────────────────────────────────────────────────────────────────

function buildExtractionPrompt(
  documentText: string,
  filename: string,
  fileType: string
): string {
  return `You are helping someone understand a document they just uploaded.
Your job: pull out the most useful things from it.

Document name: "${filename}"
Document type: ${fileType}

---

DOCUMENT CONTENT:
${documentText.slice(0, 50000)}
${documentText.length > 50000 ? '\n[Document continues — analysing first 50,000 characters]' : ''}

---

Extract the following. Be specific — only include what is actually in the document.
Do NOT invent or guess. If something doesn't exist in the document, return empty array.

Return ONLY valid JSON. No explanation. No markdown. Just the JSON object.

{
  "summary": [
    "One clear sentence about what this document covers",
    "Second key point",
    "Third key point",
    "Fourth key point"
  ],

  "tasks": [
    "Complete Organic Chemistry revision",
    "Submit assignment by Friday",
    "Call Dr. Sharma about test results"
  ],

  "dates": [
    {
      "label": "Chapter test",
      "dateText": "July 20, 2026",
      "isoDate": "2026-07-20T00:00:00"
    },
    {
      "label": "Night duty",
      "dateText": "June 18, 8 PM",
      "isoDate": "2026-06-18T20:00:00"
    }
  ],

  "examQuestions": [
    "What is double fertilisation and where does it occur?",
    "Differentiate between mitosis and meiosis.",
    "Explain the role of DNA polymerase in replication."
  ],

  "terms": [
    "Mitosis — cell division producing two identical daughter cells",
    "Meiosis — cell division producing four genetically unique cells"
  ],

  "isStudyContent": true,

  "hasDates": true,

  "hasTasks": true
}

RULES:
→ summary: 3-6 items. One sentence each. Clear and specific to this document.
→ tasks: Only real action items that need to be done. Ignore headings.
→ dates: Only actual dates/times mentioned. Parse to ISO if year is clear.
→ examQuestions: Only for textbooks, notes, coaching material.
   If not study content → return empty array.
→ terms: Only important definitions explicitly stated in the document.
→ isStudyContent: true if this is a textbook, notes, exam prep material.
→ hasDates: true if dates array is not empty.
→ hasTasks: true if tasks array is not empty.
`;
}


// ─────────────────────────────────────────────────────────────────────────────
// THE MAIN FUNCTION
// Call this after extractDocument() returns the text
// ─────────────────────────────────────────────────────────────────────────────

export async function extractDocumentItems(
  documentText: string,
  filename: string,
  fileType: string,
  openai: any  // your existing OpenAI client
): Promise<DocumentExtracts> {

  const empty: DocumentExtracts = {
    summary: [],
    tasks: [],
    dates: [],
    examQuestions: [],
    terms: [],
    isStudyContent: false,
    hasDates: false,
    hasTasks: false,
  };

  if (!documentText || documentText.trim().length < 50) {
    return empty;
  }

  try {
    const response = await openai.chat.completions.create({
      model: 'gpt-4o-mini',  // cheap + fast — perfect for extraction
      max_tokens: 1500,
      temperature: 0.1,      // low temperature = consistent structured output
      messages: [
        {
          role: 'user',
          content: buildExtractionPrompt(documentText, filename, fileType),
        },
      ],
    });

    const raw = response.choices[0]?.message?.content || '';

    // Clean up response (remove any accidental markdown)
    const cleaned = raw
      .replace(/```json/g, '')
      .replace(/```/g, '')
      .trim();

    const parsed = JSON.parse(cleaned) as DocumentExtracts;

    // Validate and sanitise
    return {
      summary:       Array.isArray(parsed.summary)       ? parsed.summary.slice(0, 6)       : [],
      tasks:         Array.isArray(parsed.tasks)         ? parsed.tasks.slice(0, 10)        : [],
      dates:         Array.isArray(parsed.dates)         ? parsed.dates.slice(0, 10)        : [],
      examQuestions: Array.isArray(parsed.examQuestions) ? parsed.examQuestions.slice(0, 5) : [],
      terms:         Array.isArray(parsed.terms)         ? parsed.terms.slice(0, 8)         : [],
      isStudyContent: Boolean(parsed.isStudyContent),
      hasDates:       Boolean(parsed.hasDates),
      hasTasks:       Boolean(parsed.hasTasks),
    };

  } catch (err) {
    console.error('Document extraction failed:', err);
    return empty;
  }
}
