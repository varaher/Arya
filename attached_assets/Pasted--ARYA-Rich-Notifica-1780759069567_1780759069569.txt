// ─── Types ────────────────────────────────────────────────────────────────────

interface MorningCardData {
  firstName: string;
  greeting: string;           // "Good morning" / "Good evening"
  kaalWindow: string;         // "10:30 AM – 12:15 PM"
  kaalEnergy: number;         // 1-10
  kaalTheme: string;          // "Communication · Expression"
  topGoal: string;            // "POCUS conference speech"
  dayName: string;            // "Wednesday"
  date: string;               // "4 Jun"
  moonSign?: string;          // "Mithun · Gemini"
}

interface GoalCheckInCardData {
  firstName: string;
  goalTitle: string;
  daysAgo: number;
  checkInType: string;        // "stalled_early" etc
}

interface StoryCardData {
  firstName: string;
  rasa: string;               // "Karuna" / "Shanta" etc
  rasaEmoji: string;          // "🌙" / "⚡" etc
  storyHint: string;          // "A tale for someone carrying a hard day"
  language: string;           // "Malayalam" / "Hindi"
}

interface ReminderCardData {
  title: string;
  time: string;               // "8:00 PM"
  isAlarm: boolean;
  message?: string;
}

interface SundayCardData {
  firstName: string;
  weekSummary: string;        // "A week of focus and quiet progress"
  goalsActive: number;
  goalsMoved: number;
}

// ─── SVG Templates ────────────────────────────────────────────────────────────

function escapeXml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;');
}

function truncate(str: string, maxLen: number): string {
  if (str.length <= maxLen) return str;
  return str.slice(0, maxLen - 3) + '...';
}

// ── 1. Morning KAAL Briefing Card ─────────────────────────────────────────────

  const energyWidth = Math.round(energyPct * 4.2); // out of 420px bar

    data.kaalEnergy >= 8 ? '#f59e0b' :
    data.kaalEnergy >= 6 ? '#4ade80' :
    data.kaalEnergy >= 4 ? '#60a5fa' :
    '#6b7280';

  const energyLabel =
    data.kaalEnergy >= 8 ? 'Peak clarity' :
    data.kaalEnergy >= 6 ? 'High energy' :
    data.kaalEnergy >= 4 ? 'Moderate' :

  const themeText = truncate(escapeXml(data.kaalTheme), 36);

  const accentColor = isAlarm ? '#ef4444' : '#4ade80';
  const bgColor1 = isAlarm ? '#1a0505' : '#051a0a';

  const badgeColor = isAlarm ? '#f87171' : '#4ade80';

  return `<svg xmlns="http://www.w3.org/2000/svg" width="600" height="220" viewBox="0 0 600 220">
  <defs>

      <stop offset="100%" style="stop-color:${bgColor2};stop-opacity:1"/>
    </linearGradient>
  </defs>

  <rect width="600" height="220" fill="none" stroke="${accentColor}" stroke-opacity="0.1" stroke-width="1" rx="16"/>
  <rect x="0" y="0" width="3" height="220" fill="${accentColor}" opacity="0.7" rx="2"/>

  <!-- Badge -->

// ─── SVG to Data URL ──────────────────────────────────────────────────────────
// Converts SVG string to a base64 data URL for use in push notifications

export function svgToDataUrl(svg: string): string {
  const base64 = Buffer.from(svg).toString('base64');
  return `data:image/svg+xml;base64,${base64}`;
}

// For serving as a URL endpoint instead of data URL:
export function svgToBuffer(svg: string): Buffer {
  return Buffer.from(svg, 'utf-8');
}

// ─── Card Dispatcher ──────────────────────────────────────────────────────────

export type CardType =
  | 'morning'
  | 'goal_checkin'
  | 'story'
  | 'reminder'
  | 'alarm'
  | 'sunday_review';

export function generateCard(type: CardType, data: any): string {
  switch (type) {
    case 'morning':      return generateMorningCard(data);
    case 'goal_checkin': return generateGoalCheckInCard(data);
    case 'story':        return generateStoryCard(data);
    case 'reminder':     return generateReminderCard({ ...data, isAlarm: false });
    case 'alarm':        return generateReminderCard({ ...data, isAlarm: true });
    case 'sunday_review':return generateSundayCard(data);
    default:             return generateMorningCard(data);
  }
}
