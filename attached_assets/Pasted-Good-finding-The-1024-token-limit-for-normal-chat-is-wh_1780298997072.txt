## The Fix — Smarter Token Detection

Instead of one limit for all chat — detect what the user is asking for:

```typescript
// server/arya/chat-engine.ts
// Find the max_tokens setting

function getMaxTokens(
  message: string,
  mode: string,
  userPref: string
): number {

  const msg = message.toLowerCase();

  // Voice — always short
  if (mode === 'voice') return 300;

  // Long form content detection
  const isLongForm =
    msg.includes('article') ||
    msg.includes('essay') ||
    msg.includes('speech') ||
    msg.includes('write a') ||
    msg.includes('draft') ||
    msg.includes('report') ||
    msg.includes('summary') ||
    msg.includes('explain in detail') ||
    msg.includes('comprehensive') ||
    msg.includes('full') ||
    msg.includes('complete');

  if (isLongForm) return 4096; // ~3000 words

  // Deep mode
  if (mode === 'deep') return 2048;

  // User preference
  if (userPref === 'detailed') return 2048;
  if (userPref === 'concise') return 800;

  // Default normal chat
  return 1500; // was 1024, now allows fuller answers
}
```
