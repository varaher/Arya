// PART 5: The 5 contextual questions — full logic
// Time-aware, pulls from real KAAL data
// ─────────────────────────────────────────────────────────────────────────────

export function buildKaalQuestion(kaalContext: {
  todayEnergy?: number;
  todayWindow?: string;
  todayTheme?: string;
  moonSign?: string;
}): string {
  const hour = new Date().getHours();
  const energy = kaalContext.todayEnergy || 7;
  const window = kaalContext.todayWindow || 'this morning';
  const theme = kaalContext.todayTheme || 'clarity';

  // Early morning — plan the day
  if (hour >= 5 && hour < 9) {
    return `My KAAL peak today is ${window}. How should I use it — what's the most important thing to put in that window?`;
  }

  // Morning — in the peak
  if (hour >= 9 && hour < 12) {
    if (energy >= 8) {
      return `KAAL says today is a high-energy day with theme "${theme}". I'm in my peak window right now — what should I be doing?`;
    }
    return `My KAAL window is ${window} and theme is "${theme}". Based on this — how should I spend the next hour?`;
  }

  // Afternoon — past peak
  if (hour >= 12 && hour < 16) {
    return `My KAAL peak has passed for today. Theme is "${theme}". Based on my timing — what's the best use of my afternoon?`;
  }

  // Evening — wind down
  if (hour >= 16 && hour < 20) {
    return `Looking at my KAAL for today — the theme was "${theme}". What should I close out before the day ends?`;
  }

  // Night — reflect and prepare
  return `Based on my KAAL — today was a ${energy >= 7 ? 'strong' : 'quiet'} day with theme "${theme}". What should I carry into tomorrow?`;
}
