## Root Cause — One Clear Problem

Looking at the voice route code shared earlier:

```typescript
// Current voice route logic:

if (useSarvam) {
  const sttResult = await sarvamSpeechToText(audio, "unknown");
  userTranscript = sttResult.transcript;
  detectedLanguage = sttResult.languageCode; // "ml-IN"

  if (isIndianLanguage(detectedLanguage)) {
    // Translates Malayalam → English
    const translation = await sarvamTranslate(
      userTranscript, detectedLanguage, "en-IN"
    );
    queryForArya = translation.translatedText; // English text
  }
}

// ARYA gets English text → thinks in English
// BUT no language instruction given
// So ARYA responds in English

const { stream } = await generateAryaResponse(
  queryForArya,   // English
  history,
  tenantId,
  conversationId,
  userId
  // ← NO LANGUAGE INSTRUCTION
);

// Then response translated back to Malayalam
// BUT if translation fails → English stays
// AND TTS may fail on translated text
```

## The Fix — Add To Voice Route Right Now

```typescript
// server/routes.ts — voice route
// Find generateAryaResponse call
// Add language instruction

// Step 1 — after detecting language:
const detectedLang = sttResult.languageCode || "en-IN";
const langCode = detectedLang.split('-')[0]; // "ml" from "ml-IN"

// Step 2 — build instruction:
const langInstruction = getLanguageInstruction(
  langCode,
  (req as any).userName || 'friend'
);

// Step 3 — pass to ARYA:
const { stream, meta } = await generateAryaResponse(
  queryForArya,
  history,
  tenantId,
  conversationId,
  userId,
  langInstruction  // ← ADD THIS LINE
);
```

## For Hindi Crash — Specific Fix

```typescript
// Hindi was crashing because:
// 1. Sarvam translates Hindi → English
// 2. ARYA responds in English
// 3. Code tries to translate English → Hindi
// 4. Something in that chain fails → crash

// Fix: add safety guard
try {
  if (isIndianLanguage(detectedLanguage)) {
    const translatedResponse = await sarvamTranslate(
      fullResponse,
      "en-IN",
      detectedLanguage
    );
    // TTS on translated text
    const ttsResult = await sarvamTextToSpeech(
      translatedResponse.translatedText.slice(0, 500),
      detectedLanguage as SarvamLanguageCode
    );
    // send audio
  }
} catch (err: any) {
  console.error('[voice] Translation/TTS failed:', err.message);
  // Don't crash — just send text without audio
  // User gets text response at minimum
}
```
