// ── THE CORE DRISHYA PROMPT BUILDER ──────────────────────────
// This replaces or extends your existing story generation prompt.
// Wire this into generateDrishyaStory().

export function buildDrishyaStoryPrompt(
  world: 'night' | 'film' | 'everyday',
  request: string,
  language: string,
  wisdomSeed?: string,  // from wisdom-retriever, optional
): string {

  const langName = DRISHYA_LANGUAGE_NAMES[language]
    || DRISHYA_LANGUAGE_NAMES['en'];

  const register = DRISHYA_LITERARY_REGISTER[language]
    || DRISHYA_LITERARY_REGISTER['en'];

  const worldTone: Record<string, string> = {
    night: `Warm, intimate, Indian settings, quiet wisdom.
            The feeling of a story told at the end of a long day.
            Characters are ordinary people in recognisable situations.
            The wisdom arrives gently, like something noticed rather than taught.`,

    film:  `Cinematic, character-driven, dramatic but grounded.
            A scene with momentum. Characters with clear wants and obstacles.
            The wisdom is embedded in what characters do, not what they say.
            More tension than the night world. Resolution that feels earned.`,

    everyday: `The extraordinary inside the ordinary.
               A moment from daily life — a conversation, a chore,
               a commute, a meal — that cracks open to reveal something
               larger. No magical elements. Just real life, fully attended to.
               The wisdom is in the attention itself.`,
  };

  return `You are writing an original story for ARYA's Drishya feature.

LANGUAGE: Write entirely in ${langName}.
Not as a translation from English.
As if this story was originally composed in ${langName},
by someone who has lived inside this language their whole life.

LITERARY REGISTER:
${register}

WORLD: ${world.toUpperCase()}
${worldTone[world]}

CRITICAL RULES — read carefully:

1. COMPLETELY ORIGINAL.
   No recognisable characters from known texts.
   No obviously named figures from mythology or history.
   Create entirely new characters with new names.
   New settings. New situations.
   The tradition lives in the wisdom, not the surface.

2. INVISIBLE SOURCE.
   ${wisdomSeed
     ? `The following insight is the invisible skeleton of this story.
        Never state it directly. Let the events of the story make it felt.
        The reader should feel the truth arrive, not be told it.
        The wisdom seed (never reveal this): "${wisdomSeed}"`
     : `The wisdom should emerge from the story naturally.
        Never preach. Never explain the lesson.
        Let the reader discover it.`
   }

3. LENGTH AND SHAPE.
   300-400 words. A complete arc.
   Beginning that places us somewhere specific.
   Middle with genuine movement.
   Ending that resolves without explaining.

4. THE CLOSING QUESTION.
   The final line is one question, in italics.
   Not rhetorical. Genuinely open.
   Something the reader will carry into the night.
   It should feel like the story whispering to them
   directly — as if it knows something about them.

5. NO CULTURAL LABELS.
   Do not say "in ancient India" or "long ago in the East"
   or anything that marks this as a retelling.
   It is not a retelling. It is an original story.
   It happens somewhere real and specific —
   a kitchen in Chennai, a hospital corridor in Kochi,
   a train platform in Howrah, a field in Punjab —
   or somewhere timeless and unnamed.

The request from the user: "${request}"

Write the story now. Only the story. No preamble.`;
}

// ── HOW TO WIRE INTO EXISTING generateDrishyaStory() ─────────

/*
In your existing server/arya/drishya.ts,
find the generateDrishyaStory() function.

BEFORE (likely something like):
  const prompt = `Write a ${world} world story about: ${request}`;

AFTER:
  import {
    buildDrishyaStoryPrompt,
    DRISHYA_VOICE_BY_LANGUAGE,
    DRISHYA_WORLD_NAMES,
  } from './drishya-language-patch';
  import { retrieveStoryWisdom } from './wisdom-retriever';

  async function generateDrishyaStory(
    world: 'night' | 'film' | 'everyday',
    request: string,
    language: string,    // ← must be passed in from the route
    userId: number,
  ) {
    // 1. Get wisdom seed (optional — story works without it)
    const wisdomEntry = await retrieveStoryWisdom(
      world, request, language, userId
    ).catch(() => null);

    // 2. Build the full prompt
    const prompt = buildDrishyaStoryPrompt(
      world,
      request,
      language,
      wisdomEntry?.storySeed,
    );

    // 3. Generate story (your existing GPT-4o call, unchanged)
    // Just replace the old prompt with the new one above.

    // 4. For TTS narration — use Drishya-specific voice
    const voiceConfig = DRISHYA_VOICE_BY_LANGUAGE[language]
      || DRISHYA_VOICE_BY_LANGUAGE['en'];

    // Pass voiceConfig.speaker and voiceConfig.speed
    // to your Sarvam bulbul:v3 TTS call for this story.
  }
*/

    const userId = req.user.id;
