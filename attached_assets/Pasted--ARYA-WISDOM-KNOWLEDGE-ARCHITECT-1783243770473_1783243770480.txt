/*
// server/arya/wisdom-retriever.ts

import { db } from '../db';
import { arya_knowledge } from '../../shared/schema';
import { arrayOverlaps, eq, and } from 'drizzle-orm';

interface WisdomContext {
  situationTags: string[];   // from situation classifier
  emotionalState: string;    // current rasa
  gunaState: string;         // current guna
  userId: number;            // to avoid repeating same wisdom
  language: string;          // user's language
}

interface WisdomEntry {
  principle: string;         // arya_principle in user's language
  storySeed?: string;        // for Drishya if relevant
  rasa: string;
  confidence: string;
}

export async function retrieveRelevantWisdom(
  context: WisdomContext
): Promise<WisdomEntry | null> {

  // Find matching wisdom entries
  const candidates = await db
    .select()
    .from(arya_knowledge)
    .where(
      and(
        // Must match at least one situation tag
        arrayOverlaps(
          arya_knowledge.situationTags,
          context.situationTags
        ),
        // Must match emotional state or guna
        arrayOverlaps(
          arya_knowledge.emotionalTags,
          [context.emotionalState]
        ),
        // Only use reviewed, high-confidence entries
        eq(arya_knowledge.reviewed, true),
      )
    )
    .limit(5);

  if (!candidates.length) return null;

  // Score candidates by relevance
  const scored = candidates.map(entry => ({
    entry,
    score: calculateRelevanceScore(entry, context),
  }));

  scored.sort((a, b) => b.score - a.score);
  const best = scored[0].entry;

  // Get the principle in the right language
  const variants = best.languageVariants as Record<string, string>;
  const principle = variants?.[context.language] || best.aryaPrinciple;

  return {
    principle,
    storySeed: best.aryaStorySeed || undefined,
    rasa: best.rasa,
    confidence: best.confidenceLevel,
  };
}

function calculateRelevanceScore(
  entry: any,
  context: WisdomContext
): number {
  let score = 0;

  // Situation tag overlap
  const situationOverlap = entry.situationTags?.filter(
    (t: string) => context.situationTags.includes(t)
  ).length || 0;
  score += situationOverlap * 3;

  // Emotional tag match
  if (entry.emotionalTags?.includes(context.emotionalState)) score += 5;

  // Guna match
  if (entry.gunaRelevance?.includes(context.gunaState)) score += 4;

  // Rasa alignment
  if (entry.rasa === mapEmotionToRasa(context.emotionalState)) score += 3;

  return score;
}

function mapEmotionToRasa(emotion: string): string {
  const mapping: Record<string, string> = {
    'love': 'shringara',
    'joy': 'hasya',
    'grief': 'karuna',
    'sadness': 'karuna',
    'anger': 'raudra',
    'courage': 'vira',
    'fear': 'bhayanaka',
    'anxiety': 'bhayanaka',
    'disgust': 'bibhatsa',
    'wonder': 'adbhuta',
    'peace': 'shanta',
    'confusion': 'bhayanaka',
    'emptiness': 'karuna',
  };
  return mapping[emotion] || 'shanta';
}
*/

// PART 6: HOW ARYA DECIDES WHEN TO USE WISDOM
// The most important rule — wisdom used wrongly
// becomes noise, not insight
// ──────────────────────────────────────────────────────────────

export const WISDOM_DEPLOYMENT_RULES = `

## WHEN ARYA SURFACES WISDOM

Wisdom is surfaced in exactly three situations:

### 1. The user has shared something genuine
   They have opened a door — described a real
   struggle, asked a real question, sat with
   something hard. The conversation has depth.
   
   NOT when: they ask a quick factual question,
   need a task done, or are in a light exchange.

### 2. The wisdom is genuinely relevant
   Not thematically adjacent — directly relevant
   to what this specific person is experiencing
   right now, with a confidence score above 7/10.
   
   NOT when: the connection is loose, poetic,
   or requires explanation to be meaningful.

### 3. The moment has space for it
   The conversation is in a reflective mode.
   The user is not in a hurry.
   They have asked an open question.
   
   NOT when: they need a quick answer,
   are in practical problem-solving mode,
   or have just received difficult news
   (wisdom too soon after pain is invalidating).

---

## WHEN ARYA DOES NOT SURFACE WISDOM

→ When simple presence is more valuable.
  Sometimes the right response is just:
  "That sounds really hard. Tell me more."
  Wisdom at the wrong moment is noise.

→ When the user hasn't opened a door.
  Offering wisdom unsolicited is lecturing.
  ARYA waits for the invitation.

→ When wisdom has been offered recently.
  Maximum one wisdom moment per conversation.
  More than that and ARYA starts to feel
  like a fortune cookie dispenser.

→ When the user is in crisis.
  Crisis needs presence, not insight.
  Wisdom comes after the person feels heard —
  never before.

→ When ARYA isn't sure it's right.
  If the confidence score is below 7/10,
  ARYA stays with the conversation.
  A wrong wisdom lands worse than no wisdom.

---

## HOW ARYA DELIVERS WISDOM

NEVER like this:
  "According to Indian tradition..."
  "The ancient texts say..."
  "There's a concept called..."
  "As the Gita teaches..."
  "Chanakya once said..."

ALWAYS like this:
  "Something I keep coming back to..."
  "One question worth sitting with..."
  Just saying it directly as ARYA's thought.
  Or weaving it into a question:
  "What if the real question isn't X but Y?"
  (where Y contains the wisdom)

THE GOLD STANDARD:
  The user should never be able to tell
  whether what ARYA just said came from
  a 3,000-year-old text or from ARYA's
  own observation of their life.

  If they can tell — the delivery failed.
  If they can't — it worked.
`;

export async function generateDrishyaStory(
  world: 'night' | 'film' | 'everyday',
  request: string,
  language: string,
  userId: number,
) {

  // 1. Classify what kind of story the user needs
  const storyNeeds = await classifyStoryRequest(request);

  // 2. Retrieve a wisdom entry with a story seed
  const wisdomSeed = await retrieveRelevantWisdom({
    situationTags: storyNeeds.situations,
    emotionalState: storyNeeds.emotion,
    gunaState: storyNeeds.guna,
    userId,
    language,
  });

  // 3. Build the Drishya generation prompt
  // The story seed is the hidden skeleton
  // The original story emerges from it
  // But the user receives a completely original narrative

  const storyPrompt = `
You are writing an original story for ARYA's Drishya feature.

CRITICAL RULES:
1. This must feel completely original — not a retelling.
   The user should have no idea this draws from any tradition.
   No character names from known texts.
   No settings that obviously reference known stories.
   Create entirely new characters, names, settings.

2. The wisdom seed below is your INVISIBLE SKELETON.
   It is the insight the story must land on.
   The story is the vehicle. The insight is the destination.
   Never state the insight directly in the story.
   Let the events of the story make it felt.

3. World: ${world}
   Night — warm, intimate, Indian settings, quiet wisdom
   Film — cinematic, dramatic, character-driven
   Everyday — ordinary moments that crack open to reveal depth

4. End with one quiet question in italics.
   Not a lesson. A door that stays open.

5. Language: ${language}
   Write entirely in the user's language.
   If the language has a poetic tradition, honour it.

6. Length: 300-400 words. Complete arc. No cliffhangers.

The wisdom seed (invisible to user, shapes the story):
"${wisdomSeed?.storySeed || request}"

The user's request: "${request}"
`;

  // Generate and stream the story
}
*/

// Wisdom must be accurate before it's trusted

/*
BEFORE any wisdom entry goes to reviewed: true,
it must be verified by a subject matter expert for:

1. ACCURACY
   Does the arya_principle faithfully represent
   what the source text actually teaches?
   Has anything been distorted by translation
   or decontextualisation?

2. APPROPRIATENESS
   Is this wisdom appropriate for a general audience?
   Could it cause harm in certain mental health contexts?
   Is there any gender, caste, or community bias
   embedded in the original that needs to be
   addressed in the arya_principle?

3. TRANSLATION QUALITY
   Are the language variants accurate and natural?
   Do they capture the same nuance as the English version?
   Are they written by a native speaker of each language?

RECOMMENDED REVIEWERS:
→ Sanskrit scholar (Gita, Upanishads, Yoga Sutras)
→ Tamil scholar (Thirukkural, Sangam literature)
→ Hindi/Urdu scholar (Kabir, Mirabai, Sufi tradition)
→ Malayalam scholar (Kerala tradition, Ashtanga Hridayam)
→ Kannada scholar (Vachanas, Basavanna)
→ Buddhist studies expert (Jataka, Abhidhamma)
→ Clinical psychologist (review for mental health safety)

This review process is what makes ARYA's wisdom trustworthy.
Unreviewed wisdom is not deployed.
*/
